// Az F1-F12 billentyűkhez tartozó nyomdamintát helyezi el a rajzlapon[cite: 2]
void Nyomda(ConsoleKey fKey)
{
    int idx = fKey - ConsoleKey.F1;     // 0..11[cite: 2]
    int meret = (idx % 3) * 2 + 3;      // 3, 5 vagy 7[cite: 2]
    int tipus = idx / 3;                // 0: X, 1: +, 2: rombuszkeret, 3: négyzetkeret[cite: 2]
    int r = meret / 2;                  // Sugár/eltolás a középponttól (1, 2 vagy 3)

    // Teljes befoglaló négyzet ellenőrzése a rajzolás előtt[cite: 2]
    if (cx - r < 0 || cx + r >= RajzlapSzelesseg || cy - r < 0 || cy + r >= RajzlapMagassag)
    {
        Hiba("Kilóg a nyomda"); // Beep és változatlan rajz[cite: 1, 2]
        return;
    }

    for (int dy = -r; dy <= r; dy++)
    {
        for (int dx = -r; dx <= r; dx++)
        {
            bool kell = tipus switch
            {
                0 => Math.Abs(dx) == Math.Abs(dy),          // X: két átló[cite: 2]
                1 => dx == 0 || dy == 0,                     // Kereszt (+): középső sor és oszlop[cite: 2]
                2 => Math.Abs(dx) + Math.Abs(dy) == r,       // Rombuszkeret: négy oldalfelező között[cite: 2]
                3 => Math.Abs(dx) == r || Math.Abs(dy) == r, // Négyzetkeret: négy szélső vonal[cite: 2]
                _ => false
            };

            if (kell)
            {
                rajz[cx + dx, cy + dy] = aktKarakter;
                szinek[cx + dx, cy + dy] = aktSzin;
            }
        }
    }

    uzenet = $"{fKey} ({meret}x{meret}) kész";
}
